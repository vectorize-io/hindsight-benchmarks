from pathlib import Path
import json, gzip, hashlib, re
import numpy as np
import torch
from transformers import AutoTokenizer, AutoModel
from sentence_transformers import SentenceTransformer
from rank_bm25 import BM25Okapi
root=Path('/mnt/optane/hindsight-benchmarks/results/experiments/embedding-reranker-locomo')
f=json.loads(gzip.decompress((root/'granite/fixture.json.gz').read_bytes()))
c=f['retriever'];torch.set_num_threads(4)
m=AutoModel.from_pretrained(c['model'],revision=c['revision'],local_files_only=True,dtype=torch.float32,attn_implementation='sdpa').eval()
t=AutoTokenizer.from_pretrained(c['model'],revision=c['revision'],local_files_only=True)
def encode(texts):
 result=[]
 for offset in range(0,len(texts),32):
  inputs=t(texts[offset:offset+32],padding=True,truncation=True,max_length=8192,return_tensors='pt')
  with torch.inference_mode():
   v=torch.nn.functional.normalize(m(**inputs).last_hidden_state[:,0],dim=1)
  result.append(v.numpy())
 return np.concatenate(result)
q=f['queries'][0];group=q['group'];ids=sorted(d for d in f['corpus'] if d.startswith(group+'/'));texts=[f['corpus'][d] for d in ids]
d=encode(texts);qs=[x for x in f['queries'] if x['group']==group];v=encode([x['query'] for x in qs]);bm=BM25Okapi([re.findall(r'\w+',x.lower()) for x in texts]);matched=0;overlaps=[]
for x,vector in zip(qs,v):
 lexical=bm.get_scores(re.findall(r'\w+',x['query'].lower()));dense=d@vector
 orders=[sorted(range(len(ids)),key=lambda i:(-float(s[i]),i)) for s in [lexical,dense]]
 ranks=[{index:rank+1 for rank,index in enumerate(order)} for order in orders]
 order=sorted(range(len(ids)),key=lambda i:(-(1/(60+ranks[0][i])+1/(60+ranks[1][i])),i))
 selected=[ids[i] for i in order[:100]];matched+=selected==x['candidates'];overlaps.append(len(set(selected)&set(x['candidates']))/100)
st=SentenceTransformer(c['model'],revision=c['revision'],device='cpu',local_files_only=True,model_kwargs={'dtype':torch.float32,'attn_implementation':'sdpa'})
s=st.encode(texts[:3],prompt='',normalize_embeddings=True,show_progress_bar=False)
audit={'model':c['model'],'revision':c['revision'],'fixture_sha256':f['fixture_sha256'],'method':'Independent HF AutoModel float32 CPU CLS pooling, L2 normalization, BM25 and independently implemented RRF; compare frozen GPU candidates for every query in first conversation','group':group,'documents':len(ids),'queries':len(qs),'exact_order_matches':matched,'mean_candidate_set_overlap':sum(overlaps)/len(overlaps),'document_l2_norm_min':float(np.linalg.norm(d,axis=1).min()),'document_l2_norm_max':float(np.linalg.norm(d,axis=1).max()),'query_l2_norm_min':float(np.linalg.norm(v,axis=1).min()),'query_l2_norm_max':float(np.linalg.norm(v,axis=1).max()),'manual_vs_sentence_transformers_max_abs_error':float(np.abs(s-d[:3]).max()),'prompts':st.prompts,'pooling':'CLS from pinned model modules','script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
(root/'granite-normalization-audit.json').write_text(json.dumps(audit,indent=2)+'\n');print(json.dumps(audit,indent=2),flush=True)
